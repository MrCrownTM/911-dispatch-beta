const express = require('express');
const cors = require('cors');
const fs = require('fs');
const path = require('path');

const app = express();
const port = process.env.PORT || 3000;

app.use(cors());
app.use(express.json({ limit: '50mb' }));

const DATA_FILE = path.join(__dirname, 'data.json');

function loadData() {
    try {
        if (fs.existsSync(DATA_FILE)) {
            const data = fs.readFileSync(DATA_FILE, 'utf8');
            return JSON.parse(data);
        }
    } catch(e) {}
    return {
        calls: [],
        units: [],
        archive: [],
        logs: [],
        history: {},
        roles: {},
        config: {},
        timestamp: new Date().toISOString()
    };
}

function saveData(data) {
    fs.writeFileSync(DATA_FILE, JSON.stringify(data, null, 2));
}

let db = loadData();

// API
app.get('/api/data', (req, res) => {
    res.json(db);
});

app.post('/api/data', (req, res) => {
    try {
        const newData = req.body;
        db = { ...db, ...newData, timestamp: new Date().toISOString() };
        saveData(db);
        res.json({ success: true });
    } catch(e) {
        res.status(500).json({ error: e.message });
    }
});

app.post('/api/section/:section', (req, res) => {
    const section = req.params.section;
    if (db[section] !== undefined) {
        db[section] = req.body;
        db.timestamp = new Date().toISOString();
        saveData(db);
        res.json({ success: true });
    } else {
        res.status(404).json({ error: 'Section not found' });
    }
});

app.post('/api/log', (req, res) => {
    const log = req.body;
    if (!db.logs) db.logs = [];
    db.logs.push(log);
    if (db.logs.length > 1000) db.logs = db.logs.slice(-1000);
    saveData(db);
    res.json({ success: true });
});

app.post('/api/history/:callId', (req, res) => {
    const callId = req.params.callId;
    const entry = req.body;
    if (!db.history) db.history = {};
    if (!db.history[callId]) db.history[callId] = [];
    db.history[callId].push(entry);
    saveData(db);
    res.json({ success: true });
});

app.get('/api/history/:callId', (req, res) => {
    const callId = req.params.callId;
    res.json(db.history[callId] || []);
});

app.get('/', (req, res) => {
    res.json({
        status: 'online',
        version: '1.0.0',
        data: {
            calls: db.calls?.length || 0,
            units: db.units?.length || 0,
            archive: db.archive?.length || 0,
            logs: db.logs?.length || 0
        },
        timestamp: db.timestamp || new Date().toISOString()
    });
});

app.listen(port, '0.0.0.0', () => {
    console.log(`🚀 Сервер запущен на порту ${port}`);
});
